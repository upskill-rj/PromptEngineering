

# Problem = OAuth2 - Authorization Code Flow


```pseudocode
BEGIN

    FLOW OAuth2AuthorizationCode

        // Step 1 - Client requests authorization
        CLIENT -> AUTHORIZATION_SERVER
            GET /oauth2/authorize
            PARAMS: client_id, redirect_uri, scope, state, response_type="code"

        // Step 2 - User authenticates & consents
        AUTHORIZATION_SERVER
            AUTHENTICATE user (username + password OR SSO)
            SHOW consent screen (requested scopes)
            IF user Approves THEN
                SET authCode = generateAuthCode(userID, clientID, scope, ttl=120s)
                STORE authCode IN cache WITH expiry
                REDIRECT to redirect_uri + "?code=" + authCode + "&state=" + state
            ELSE
                REDIRECT to redirect_uri + "?error=access_denied"
            END IF

        // Step 3 - Client exchanges code for token
        CLIENT -> AUTHORIZATION_SERVER
            POST /oauth2/token
            BODY: grant_type="authorization_code",
                  code=authCode,
                  redirect_uri, client_id, client_secret
        
        AUTHORIZATION_SERVER
            VALIDATE client_id AND client_secret
            VALIDATE authCode (not expired, not used)
            MARK authCode as USED
            SET accessToken = generateJWTToken(user, scope, ttl=3600)
            SET refreshToken = generateOpaqueToken(ttl=86400)
            STORE refreshToken IN database
            RETURN { access_token, refresh_token, token_type, expires_in }

        // Step 4 - Token Refresh
        FUNCTION refreshAccessToken(refeshToken)
            LOOKUP refreshToken IN database
            IF NOT FOUND OR expired THEN THROW InvalidRefreshTokenException
            SET newAccessToken = generateJWTToken(user, scope, ttl=3600)
            ROTATE refreshToken (invalidate old, issue new)
            RETURN newAccessToken
        END FUNCTION
    END FLOW


END
```
