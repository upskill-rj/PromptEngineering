

# Problem = Containerization - Docker Build Pipeline



```pseudocode
BEGIN

    PIPELINE ContainerBuild(service)

    STAGE "Build"
        // Multi-stage Docker build
        STAGE_1 "builder"
            FROM maven:3.9-openjdk-21 AS builder
            COPY source code
            RUN mvn clean package -DskipTests
            OUTPUT: app.jar

        STAGE_2 "runtime"
            FROM eclipse-temurin:21-jre-alpine
            CREATE non-root user (uid=1001)
            COPY app.jar FRO builder
            EXPOSE 8080
            SET JVM_OPTS = "-XX:MaxRAMPercentage=75"
            ENTRYPOINT ["java", JVM_OPTS, "-jar", "app.jar"]

    STAGE "Scan"
        RUN trivy image SCAN FOR vulnerabilities
        IF CRITICAL vilnerabilities FOUND THEN
            FAIL pipeline
            NOTIFY security team
        END IF
        RUN SAST scan (SonarQube)
        RUN dependency check (OWASP)

    STAGE "Push"
        TAG image WITH gitCommitSha, version, latest
        PUSH TO containerRegistry
        SIGN image WITH cosign (supply chain security)

    STAGE "DEPLOY"
        UPDATE K8s deployment maifest WITH new image tag
        APPLY manifest VIA kubectl, OR helmUpgrade
        VERIFY rollout status
    END PIPELINE


END
```
