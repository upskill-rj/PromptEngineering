

# Problem = Observability - Metrics, Logs, Traces


```pseudocode
BEGIN

    SYSTEM ObservabilityStack

        // Structured Logging
        FUNCTION log(level, message, context)
            SET logEntry = {
                timestamp:      ISO8601(),
                level:          level,
                service:        SERVICE_NAME,
                traceID:        MDC.get("treaceID"),
                spanID:         MDC.get("correlationID"),
                userID:         context.userID,
                message:        message,
                environment:    ENV
            }
            WRITE logEntry AS JSON TO stdout        // Collected by Fluented/Filebeat
        END FUNCTION

        // Distributed Tracing (Open Telemetry)
        FUNCTION traceRequest(request)
            SET span = tracer.startSpan(request.path)
            SET span.attributes = {
                http.method:    request.method,
                http.url:       request.url,
                user.id:        request.userID
            }
            TRY
                SET result = processRequest(request)
                SET span.status = OK
                RETURN result
            CATCH error
                SET span.status = ERROR
                SET span.recordException(error)
                THROW error
            FINALLY
                span.end()
            END TRY
        END FUNCTION


        // Metrics (Micrometer -> Prometheus)
        DEFINE metrics
            COUNTER     http_requests_total [method, path, status]
            HISTOGRAM   http_request_duration_seconds [method. path]
            GAUGE       jvm_memory_used_bytes [area]
            GAUGE       db_connection_pool_active []
            COUNTER     business_orders_created_total []
        END DEFINE

        // Alerting Rules (Prometheus/Grafana)
        ALERT HighErrorRate
            IF rate(http_requests_total{status=~"5.."}[5m]) / rate(http_requests_total[5m]) > 0.05
            FOR 2 minutes
            NOTIFY: PagerDuty (P1), Slack #alerts

        ALERT HighLatency
            IF histogram_quantile(0.99, http_request_duration_seconds) > 2.0
            FOR 5 minutes
            NOTIFY: Slack #alerts

        ALERT ProdCrashLooping
            IF kube_pod_container_status_restarts_total > 5
            FOR 15 minutes
            NOTIFY: PagerDuty (P1)

    END SYSTEM

END
```
