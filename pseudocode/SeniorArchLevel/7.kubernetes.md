


# Problem = Kubernetes - Deployment & Autoscaling


```pseudocode
BEGIN

    DEFINE K8s_Deployment(UserService)

        MANIFEST Deployment
            replicas: 3
            selector: app=user-service
            template:
                image: registry/user-service:v2.1.0
                resources:
                    requests: cpu=250m, memory=256Mi
                    limits:   cpu=500m, memory=512Mi
                livenessProbe:
                    GET /actuator/health/liveness
                    initialDelay=30s, period=10s, failureThreshold=3
                readinessProbe:
                    GET /actuator/health/readiness
                    initialDelay=10s, period=5s
                envFrom:
                    secretRef: user-services-secrets
                    configMapRef: user-service-config

        MANIFEST HorizontalPodAutoscaler
            target: Deployment/user-service
            minReplicas: 3
            maxReplicas: 20
            metrics:
                cpu: targetAverageUtilization=70%
                memory: targetAverageValue=400Mi
                custom: requests_per_second > 1000
        
        MANIFEST ProdDisruptionBudget
            minAvailable: 2  // Always keep 2 pods running

        FUNCTION rollingUpdate(newImage)
            SET strategy = RollingUpdate
            SET maxSurge = 1        // 1 extra pod during update
            SET maxUnavailable = 0  // Zero Downtime
            FOR each pod IN deployment DO
                SPIN UP new pod WITH newImage
                WAIT FOR readinessProbe to PASS
                TERMINATE old pod
            END FOR
        END FUNCTION

        FUNCTION canaryDeploy(canaryImage, trafficPercent=10)
            DEPLOY canary pods WITH canaryImage
            ROUTE trafficPercent% of traffic -> canary
            MONITOR error_rate, latency FOR 30 minutes
            IF error_rate > 1% THEN
                ROLLBACK canary
            ELSE
                GRADUALLY increase canary traffic to 100%
                DECOMMISSION old pods
            ENF IF
        END FUNCTION
END
```
