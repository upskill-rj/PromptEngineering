



# Problem = Serverless - Event-Driven Functions


```pseudocode
BEGIN

    FUNCTION OderProcessor (event) [AWS Lambda / GCP Cloud Run / OCI Functions]

        ON TRIGGER FROM MessageQueue("order-events")
            SET order = PARSE(event.body)
            VALIDATE order schema
            IF invalid THEN
                SEND TO deadLetterQueue(event)
                RETURN
            END IF

            TRY
                BEGIN distributed_saga

                    STEP 1 - InventoryService
                        RESERVE inventory(order.items)
                        ON FAIL -> compensate: RELEASE reservation

                    STEP 2 - PaymentService
                        CHARGE payment(order.amount, order.customerID)
                        ON FAIL -> compensate: REFUND payment

                    STEP 3 - ShippingService
                        CREATE shipment(order.deliveryAddress)
                        ON FAIL -> compensate: CANCEL shipment

                    STEP 4 - NotificationService
                        SEND confirmation email/SMS to customer

                COMMIT saga
                PUBLISH event("order.completed", orderID) TO EventBus

            CATCH SagaFailureException
                EXECUTE all compensation steps IN REVERSE ORDER
                PUBLISH event("order.failed", orderID) TO EventBus
                LOG failure WITH full context
            END TRY
        
        CONFIGURE
            timeout:        30s
            memoryMB:       512
            concurrency:    100
            retryAttempts:  3
            dlq:            "order-events-dlq"
        END CONFIGURE
    
    END FUNCTION

END
```
