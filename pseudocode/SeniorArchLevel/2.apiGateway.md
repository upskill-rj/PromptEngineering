

# Problem = API Gateway - Routing & Filtering


```pseudocode
BEGIN

    GATEWAY ApiGateway
        CONFIGURE port = 8080
        CONFIGURE loadBalancer = "ROUND_ROBIN"

    DEFINE routes
        ROUTE "/api/users/**"    -> UserService
        ROUTE "/api/orders/**"   -> OrderService
        ROUTE "/api/payments/**" -> PaymentService
        ROUTE "/api/products/**" -> ProductService
        ...........
    END ROUTES

    DEFINE filterChain

        PRE_FILTER
            EXTRACT token FROM request.header["Authorization"]
            IF token is NULL THEN REJECT(401, "Unauthorized")
            VALIDATE token WITH JwtService
            IF token INVALID THEN REJECT(403, "Forbidden")
            ATTACH userID, roles TO request.context
        END PRE_FILTER

        POST_FILTER
            LOG request.path, response.status, response.time
            ADD header["X-Response-Time"] = elapsed
            ADD header["X-Correlation-ID"] = correlationID
        END POST_FILTER

        RATE_LIMIT_FILTER
            GET clientIP FROM request
            GET requestCount FROM redisCache[clientIP]
            IF requestCount > 1000 per minute THEN 
                REJECT(429, "Too Many Requests...")
            END IF
            INCREMENT redisCache[clientIP]
        END RATE_LIMIT_FILTER
    END FILTER_CHAIN
END GATEWAY

END
```
