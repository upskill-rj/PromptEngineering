

# Problem = OAuth2 - Authorization Code Flow


```pseudocode
BEGIN

    FLOW SingleSignOn_OIDC

        // User visits Service Provider (SP)
        USER -> SERVICE_PROVIDER("/dashboard")
            SP checks session -> NOT FOUND
            SP redirects -> IDENTITY_PROVIDER (Idp)
                PARAMS: client_id, nonce, scope="openid profile email", response_type="code", redirect_uri
        
        // Idp authenticates user
        IDENTITY_PROVIDE
            IF user has active SSO session THEN
                SKIP login form
                ISSUE authCode IMMEDIATELY
            ELSE
                SHOW login form
                AUTHENTICATE user credentials
                CREATE SSO session (set SSO cookie, ttl=8h)
                ISSUE authCode
            END IF
            REDIRECT back to SP with authCode

        // SP exchanges code -> ID token + Access token
        SERVICE_PROVIDE
            POST /token to Idp
            RECEIVE id_token (JWT with user claims)
            VALIDATE id_token (signature, nonce, expiry, issuer)
            EXTRACT sub, email, name, groups, FROM id_token
            CREATE local session FOR user
            SET session cookie (HttpOnly, Secure. SameSite=Strict)
            RENDER "/dashboard"

        // SSO Logout (Single Logout)
        FUNCTION ssoLogout(userID)
            GET allActiveSessions FOR userID FROM sessionStore
            FOR each session DO
                INVALIDATE session
                NOTIFY relying party (back-channel logout)
            END FOR
            CLEAR SSO cookie AI Idp
            REDIRECT to logout confirmation page
        END FUNCTION
        
    END FLOW

END
```
