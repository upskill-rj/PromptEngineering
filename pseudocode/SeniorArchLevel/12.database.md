



# Problem = Database - Sharding, Replication & CQRS


```pseudocode
BEGIN

    ARCHITECTURE DatabaseLayer

        // CQRS Pattern
        COMMAND_SIDE (Write)
            DATABASE:   PostgreSQL Primary (RDS/CloudSQL/OCI ADB)
            PATTERN:    All writes go to primary
            STRATEGY:   Optimistic locking (version field)
            FUNCTION writeUser (command)
                BEGIN transaction
                    SET existing = SELECT * FROM users WHERE id = command.id
                    IF existing.version != command.expectedVersion THEN THROW OptimisticLockException
                    END IF
                    UPDATE users SET ... version = version + 1
                    INSERT INTO outbox_events (event_type, payload)
                COMMIT transaction
                // Outbox pattern -> reliable event publishing
            END FUNCTION

        QUERY_SIDE (Read)
            DATABASE:   PostgreSQL Read Replicas OR Elasticsearch
            PATTERN:    All reads from replica/search index
            CACHE:      Redis (TTL-based, cache-aside)
            FUNCTION readUser(userID)
                SET cached = redisCache.get("user:" + userID)
                IF cached NOT NULL THEN RETURN cached
                SET user = readReplica.query("SELECT * FROM users WHERE id=?",userID)
                redisCache.set("user:" + userId, user, ttl=300)
                RETURN user
            END FUBCTION

        // Database Sharding
        FUNCTION getShardConnect(userID)
            SET shardCount = 8
            SET shardKey = userID MOD shardCount
            RETURN connectionPool["shard-" + shardKey]
        END FUNCTION

        // Replication Lag Monitoring
        FUNCTION monitorReplicationLag()
            SET lag = primary.lsn - replica.lsn
            IF lag > 10MB THEN
                ALERT(" Replication lag critical: " + lag)
                ROUTE read TO primary TEMPORARILY
            END IF
        END FUNCTION
    END ARCHITECTURE

END
```
