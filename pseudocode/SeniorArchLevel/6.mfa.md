

# Problem = MFA - Multi-Factor Authentication Flow


```pseudocode
BEGIN

    FLOW MultiFactorAuthentication

        FUNCTION initiateLogin(username, password)
            SET user = findUserByUsername(username)
            IF user IS NULL THEN THROW UserNotFoundException
            IF NOT verifyPassword(password, user.hashedPassword) THEN
                INCREMENT user.failedAttempts
                IF user.failedAttempts >= 5 THEN
                    LOCK user account FOR 30 minutes
                    SEND lockout alert email
                END IF
                THROW InvalidCredentialsException
            END IF
            RESET user.failedAttempts
            IF user.mfaEnabled THEN
                RETURN initiateMfaChallenge(user)
            ELSE
                RETURN generateJWTToken(User)
            END IF
        END FUNCTION

        FUNCTION initiateMFAChallenge(user)
            SWITCH user.mfaMethod
                CASE "TOP"
                    RETURN { mfaRequired: true, method: "TOTP" }

                CASE "SMS"
                    SET otp = generateOTP(length=6. ttl=300)
                    STORE otp IN cache[user.id]
                    SEND otp VIA smsGateway(user.phone)
                    RETURN { mfaRequired: true, method: "SMS" }
                CASE "MAIL"
                    set otp = generateOTP(length=6. ttl=300)
                    STROE otp IN cache[user.id]
                    SEND otp VIA emailService(user.email)
                    RETURN { mfaRequired: true, method: "EMAIL"}

                CASE "HARDWARE_KEY"
                    SET challenge = generateWebAuthnChallenge()
                    RETURN { mfaRequired: true, method: "FID02", challenge }
            END SWITCH
        END FUNCTION

        FUNCTION verifyMFA(userID, OTP, method)
            SWITCH method
                CASE "TOTP"
                    SET secret = getUser(userID).totpSecret
                    IF NOT verifyTOTP(otp, secret, window=1) THEN
                        THROW InvalidOTPException
                    END IF

                CASE "SMS" OR "EMAIL"
                    SET storedOTP = cache[userID]
                    IF storeOTP IS NULL THEN THROW OTPExpiredException
                    ID storedOTP != OTP THEN THROW InvalidOTPException
                    DELTE cache[userID]

                CASE "FID02"
                    VERIFY webAuthnAssertion(userID, OTP)
            END SWITCH

            SET user = getUser(userID)
            RETURN generateJWTToken(user)
        END FUNCTION
    END FLOW

END
```
