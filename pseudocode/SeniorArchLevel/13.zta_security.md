

# Problem = Security - Zero Trust Architecture


```pseudocode
BEGIN

    ARCHITECTURE ZeroTrustSecurity

        PRINCIPLE: "Never trust, always verify"

        FUNCTION enforceZeroTrust(request)

            // 1. Identity Verification
            VERIFY identity(JWT + MFA status)
            CHECK device posture (managed fevice, OS patch level)
            CHECK user risk score (anomaly detection)

            // 2. Network Security
            ENFORCE mTLS between ALL services
            VERIFY client certificate
            IF certificate INVALID OR expired THEN REJECT

            // 3. AUTHORIZATION (OPA - Open Policy Agent)
            SET policy = loadPolicy("rbac-policy.rego")
            SET decision = OPA.evaluate(policy, {
                user:       request.user,
                resource:   request.path,
                action:     request.method
                context:    {time, location, deviceTrust}
            })
            IF decision == DENY THEN REJECT(403)

            // 4. Data Security
            ENCRYPT data AT REST (AES-256)
            ENCRYPT data IN TRANSIT (TLS 1.3)
            MASK PII in logs (email, phone, SSN)
            TOHENIZE payment card data (PCI-DSS)

            // 5. Secrets Management (Vault / AWS Secrets Manager)
            FUNCTION getSecret(secretName)
                IF secretCache[secretName] NOT expired THEN
                    RETURN secretCache[secretName]
                END IF
                SET secret = vault.read("secret/data" + secretName)
                secretCache[secretName] = secret WITH ttl=3600
                RETURN secret
            END FUNCTION

            // 6. WAF Rules
            INSPECT request FOR
                SQL injection patterns
                XSS payloads
                CSRF token validity
                Request size > 10MB
            IF threat DETECTED THEN
                BLOCK request
                LOG threat WITH IP, payload
                INCREMENT IP reputation score
            END IF
        END FUNCTION

        // Security Scanning Pipeline
        PIPELINE SecurityChecks
            RUN SAST -> SonarQube (code vulnerabilities)
            RUN DAST -> OWASP ZAP (runtime vulnerabilities)
            RUN SCA -> Snyk (dependency CVEs)
            RUN IaC -> Checkov (Terraform / K8s misconfigurations)
            RUN container scan -> Trivy
            IF any CRITICAL finding THEN BLOCK deployment
        END PIPELINE
    END ARCHITECTURE
END
```
