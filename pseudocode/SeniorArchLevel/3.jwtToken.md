

# Problem = JWT - Token Generation & Validation


```pseudocode
BEGIN

    FUNCTION generateJWTToken(user)
        SET header = {
            alg: "RS256",
            typ: "JWT"
        }
        SET payload = {
            sub:    user.id,
            email:  user.email,
            roles:  user.roles,
            iss:    "auth-service",
            aud:    "api-gateway",
            iat:    currentTimestamp(),
            exp:    currentTimestamp() + 3600,
            jti:    generateUUID()  //prevent replay attacks
        }
        SET privateKey = loadFromVault("jwt-private-key")
        SET token = BASE64(header) + "." + BASE64(payload)
        SET signature = RS256_SIGN(token, privateKey)
        RETURN token + "." + signature
    END FUNCTION

    FUNCTION validateJWTToken(token)
        TRY
            SET parts = SPLIT(token, ".")
            IF LENGTH(parts) != 3 THEN THROW InvalidTokenException

            SET publicKey = loadFromVault("jwt-public-key")
            VERIFY signature USING publicKey
            SET payload = DECODE(parts[1])

            IF payload.exp < currrentTimestamp() THEN 
                THROW TokenExpiredException
            END IF
            IF payload.iss != "auth-service" THEN
                THROW InvalidIssuerException
            END IF
            IF isTokenBlacklisted(payload.jti) THEN
                THROW RevokedTokenException
            END IF

            RETURN payload
        CATCH exception
            LOG "Token validation failed: " + exception.message
            RETURN NULL
        END TRY

    END FUNCTION

END
```
