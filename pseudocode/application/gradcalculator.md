
# Problem = Grade Calculator


```pseudocode
BEGIN

    READ marks

    IF marks >= 90 THEN
        PRINT "GRADE A"
    ELSE IF marks >= 75 THEN
        PRINT "GRADE B"
    ELSE IF marks >= 60 THEN
        PRINT "GRADE C"
    ELSE IF marks >= 40 THEN
        PRINT "GRADE D"
    ELSE
        PRINT "GRADE F - Fail"
    END IF

END
```
